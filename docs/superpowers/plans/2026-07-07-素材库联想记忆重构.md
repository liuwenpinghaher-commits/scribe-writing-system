```python
# 用真实 SurrealDB(测试库),不 mock DB
def test_jieba_bm25_chinese_ranking(store):
    import jieba
    docs = [("血玉铜炉在密室中散发寒气", "d1"), ("江湖上流传的传说", "d2"),
            ("密室里的机关暗器", "d3")]
    for text, did in docs:
        toks = " ".join(jieba.cut(text))
        store._execute_write("CREATE type::thing('probe',$id) SET tokens=$t",
                             {"id": did, "t": toks})
    store._execute_write("DEFINE ANALYZER probe_zh TOKENIZERS blank FILTERS lowercase", {})
    store._execute_write("DEFINE INDEX probe_fts ON probe FIELDS tokens SEARCH ANALYZER probe_zh BM25", {})
    q = " ".join(jieba.cut("密室"))
    rows = store._query(
        "SELECT meta::id(id) AS id, search::score(1) AS s FROM probe WHERE tokens @@ $q ORDER BY s DESC",
        {"q": q})
    ids = [r["id"] for r in rows]
    assert "d1" in ids and "d3" in ids  # 含"密室"的命中
    assert "d2" not in ids              # 不含的不命中
```

```python
def add_material_reference(self, material_id, chapter, dimension, reason, via="manual"):
    self._execute_write(
        "CREATE material_reference SET material_id=$m, chapter=$c, dimension=$d, reason=$r, via=$v",
        {"m": material_id, "c": chapter, "d": dimension, "r": reason, "v": via})
    self._execute_write(
        "UPDATE type::record('material',$m) SET recall_count += 1, last_recalled_at = time::now()",
        {"m": material_id})

def material_refs_by_material(self, material_id):
    return self._query("SELECT * FROM material_reference WHERE material_id=$m ORDER BY chapter", {"m": material_id})
def material_refs_by_chapter(self, chapter):
    return self._query("SELECT * FROM material_reference WHERE chapter=$c", {"c": chapter})
```
