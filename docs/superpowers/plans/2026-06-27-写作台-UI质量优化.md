```jsx
import { useEffect, useRef, useState } from 'react'
import { Btn } from '../ui.jsx'

export function Dialog({ open, onClose, children, labelledBy }) {
  const ref = useRef(null)
  useEffect(() => {
    if (!open) return
    const onKey = (e) => { if (e.key === 'Escape') onClose?.() }
    document.addEventListener('keydown', onKey)
    ref.current?.focus()
    return () => document.removeEventListener('keydown', onKey)
  }, [open, onClose])
  if (!open) return null
  return (
    <div className="fixed inset-0 z-[100] grid place-items-center bg-black/50 motion-safe:animate-[fadeIn_.15s_ease-out]"
         onClick={onClose}>
      <div ref={ref} tabIndex={-1} role="dialog" aria-modal="true" aria-labelledby={labelledBy}
           onClick={(e) => e.stopPropagation()}
           className="min-w-[320px] max-w-[440px] w-[90vw] rounded-xl bg-surface border border-line shadow-[var(--shadow-card)] outline-none">
        {children}
      </div>
    </div>
  )
}

export function InputDialog({ open, title, placeholder = '', defaultValue = '', confirmText = '确认', onConfirm, onClose }) {
  const [v, setV] = useState(defaultValue)
  useEffect(() => { if (open) setV(defaultValue) }, [open, defaultValue])
  const submit = () => { onConfirm?.(v); onClose?.() }
  return (
    <Dialog open={open} onClose={onClose} labelledBy="dlg-title">
      <div className="p-4">
        <div id="dlg-title" className="text-[14px] font-medium text-ink mb-3">{title}</div>
        <input autoFocus value={v} placeholder={placeholder}
          onChange={(e) => setV(e.target.value)}
          onKeyDown={(e) => { if (e.key === 'Enter') submit() }}
          className="w-full h-9 rounded-lg border border-line bg-canvas px-3 text-[13px] text-ink outline-none focus-visible:ring-2 focus-visible:ring-brand/40 focus:border-brand" />
        <div className="mt-4 flex justify-end gap-2">
          <Btn variant="ghost" onClick={onClose}>取消</Btn>
          <Btn variant="primary" onClick={submit}>{confirmText}</Btn>
        </div>
      </div>
    </Dialog>
  )
}

export function ConfirmDialog({ open, title, body, danger, confirmText = '确认', onConfirm, onClose }) {
  return (
    <Dialog open={open} onClose={onClose} labelledBy="cfm-title">
      <div className="p-4">
        <div id="cfm-title" className="text-[14px] font-medium text-ink mb-1.5">{title}</div>
        {body && <div className="text-[12.5px] text-faint leading-relaxed mb-4">{body}</div>}
        <div className="mt-2 flex justify-end gap-2">
          <Btn variant="ghost" onClick={onClose}>取消</Btn>
          <Btn variant={danger ? 'danger' : 'primary'} onClick={() => { onConfirm?.(); onClose?.() }}>{confirmText}</Btn>
        </div>
      </div>
    </Dialog>
  )
}
```
> 若 `Btn` 无 `variant="danger"`,在 Task 12 统一补;此处临时用 `variant="primary"` + `className="!bg-danger"` 兜底,并在报告标注。

```jsx
import { createContext, useContext, useReducer, useCallback } from 'react'

export function toastReducer(state, action) {
  if (action.type === 'add') return [...state, { id: action.toast.id ?? (state.at(-1)?.id ?? 0) + 1, ...action.toast }]
  if (action.type === 'remove') return state.filter((t) => t.id !== action.id)
  return state
}

const ToastCtx = createContext(null)
export const useToast = () => useContext(ToastCtx)

export function ToastProvider({ children }) {
  const [toasts, dispatch] = useReducer(toastReducer, [])
  const toast = useCallback((msg, opts = {}) => {
    const id = Date.now() + Math.floor(Math.random() * 1000)
    dispatch({ type: 'add', toast: { id, msg, tone: opts.tone || 'ink' } })
    setTimeout(() => dispatch({ type: 'remove', id }), opts.duration ?? 4000)
  }, [])
  return (
    <ToastCtx.Provider value={toast}>
      {children}
      <div aria-live="polite" className="fixed bottom-4 left-1/2 -translate-x-1/2 z-[120] flex flex-col items-center gap-2 pointer-events-none">
        {toasts.map((t) => (
          <div key={t.id} role="status"
               className={`pointer-events-auto rounded-lg px-3 py-2 text-[12.5px] shadow-[var(--shadow-card)] border motion-safe:animate-[slideUp_.18s_ease-out] ${t.tone === 'danger' ? 'bg-danger/10 border-danger/30 text-danger' : 'bg-ink text-canvas border-transparent'}`}>
            {t.msg}
          </div>
        ))}
      </div>
    </ToastCtx.Provider>
  )
}
```
> `Date.now()/Math.random()` 在浏览器运行时可用(非 workflow 脚本环境),此处合规。
