```python
def promise_window_state(p: dict, max_chapter: int) -> dict:
    """伏笔窗口计算态。next_ch = max_chapter+1(即将写的下一章)。
    state ∈ inactive / pre_window / in_window / over_half / overdue。
    salience_gap = 距上次露面(埋设或触及)的章数,供'先刷新再兑现'门槛用。"""
    setup = p.get("setup_chapter")
    if p.get("status") != "open" or p.get("kind") == "red_herring" or setup is None:
        return {"state": "inactive", "salience_gap": None}
    next_ch = max_chapter + 1
    pmin = p.get("payoff_min") or setup + 1
    pmax = p.get("payoff_max")  # None → book 层,至 climax,永不 overdue
    last_seen = max(setup, p.get("last_refresh_at") or setup)
    gap = max_chapter - last_seen
    if next_ch < pmin:
        state = "pre_window"
    elif pmax is None:
        state = "in_window"
    elif next_ch > pmax:
        state = "overdue"
    elif next_ch > pmin + (pmax - pmin) / 2:
        state = "over_half"
    else:
        state = "in_window"
    return {"state": state, "salience_gap": gap}
```

```python
def add_promise_touch(self, promise_id: str, chapter: int,
                      evidence: str = "", ingest_seq: int | None = None) -> None:
    self._execute_write(
        "CREATE promise_touch SET promise_id=$p, chapter=$c, evidence=$e"
        + (", ingest_seq=$iseq" if ingest_seq is not None else ""),
        {"p": promise_id, "c": chapter, "e": evidence or None,
         **({"iseq": ingest_seq} if ingest_seq is not None else {})})

def list_promise_touches(self, promise_id: str) -> list[dict]:
    return self._query(
        "SELECT promise_id, chapter, evidence FROM promise_touch "
        "WHERE promise_id=$p ORDER BY chapter ASC", {"p": promise_id})

def promises_with_lifecycle(self, max_chapter: int | None = None) -> list[dict]:
    """list_promises + 派生 refresh_count/last_refresh_at(自 promise_touch)+ window 计算态。"""
    if max_chapter is None:
        max_chapter = self.max_chapter()
    touches = self._query(
        "SELECT promise_id, chapter FROM promise_touch ORDER BY chapter ASC")
    by_pid: dict[str, list[int]] = {}
    for t in touches or []:
        by_pid.setdefault(t["promise_id"], []).append(t["chapter"])
    out = []
    for p in self.list_promises():
        chs = by_pid.get(p["id"], [])
        p = dict(p, refresh_count=len(chs), last_refresh_at=(chs[-1] if chs else None))
        p["window"] = promise_window_state(p, max_chapter)
        out.append(p)
    return out
```

```python
def _resolve_promise(store, text: str) -> str | None:
    """按原文精确匹配 open 伏笔,返回 promise id;匹配不到返回 None(同 _resolve_motif 范式)。"""
    needle = (text or "").strip()
    if not needle:
        return None
    for p in store.open_promises():
        if (p.get("text") or "").strip() == needle:
            return p.get("id")
    return None
```

```python
def set_char_anchors(self, character_id: str, anchors: dict) -> None:
    sets, vars_ = [], {}
    for k in _ANCHOR_KEYS:
        if k in anchors:
            sets.append(f"{k}=${k}")
            vars_[k] = anchors[k]
    if not sets:
        return
    tbl, local = _split(character_id)
    vars_["rid"] = RecordID(tbl, local)
    self._execute_write(f"UPDATE $rid SET {', '.join(sets)}", vars_)

def char_anchors(self, character_id: str) -> dict:
    tbl, local = _split(character_id)
    rows = self._query("SELECT ghost, lie, truth, want, need, arc_type, private_agenda "
                       "FROM $rid", {"rid": RecordID(tbl, local)})
    return rows[0] if rows else {}

def add_arc_beat(self, character_id: str, beat_name: str, target_chapter: int,
                 lie_belief_target: int | None = None, trigger_event: str = "") -> None:
    self._execute_write(
        "CREATE arc_plan SET character_id=$c, beat_name=$b, target_chapter=$t, "
        "lie_belief_target=$lb, trigger_event=$te",
        {"c": character_id, "b": beat_name, "t": target_chapter,
         "lb": lie_belief_target, "te": trigger_event or None})

def list_arc_plan(self, character_id: str) -> list[dict]:
    return self._query(
        "SELECT meta::id(id) AS id, character_id, beat_name, target_chapter, "
        "lie_belief_target, trigger_event FROM arc_plan "
        "WHERE character_id=$c ORDER BY target_chapter ASC", {"c": character_id})

def delete_arc_beat(self, beat_id: str) -> None:
    self._execute_write("DELETE type::record('arc_plan', $b)", {"b": beat_id})

def latest_lie_belief(self, character_id: str, upto_chapter: int) -> int | None:
    rows = self._query(
        "SELECT lie_belief FROM char_state WHERE character_id=$c AND chapter<=$ch "
        "AND lie_belief IS NOT NONE ORDER BY chapter DESC LIMIT 1",
        {"c": character_id, "ch": upto_chapter})
```

```python
"""章前组装器:把弧光/伏笔/开环/心跳组装成「本章创作简报」,注入写作 system prompt。
设计:设计文档/2026-07-03-工艺引擎强化-design.md 第二节。
纯函数生成各 section(可测不连库);assemble_chapter_brief 是唯一连 store 的薄壳。"""
from __future__ import annotations

_TIER_W = {"book": 3, "arc": 2, "chapter": 1}
_SALIENCE_GAP_MAX = 3   # 距上次露面超过此章数 → 兑现前先刷新
_LOOP_CAP = 12          # 开环加权上限


def arc_directive(name: str, anchors: dict, arc_plan: list[dict],
                  lie_belief: int | None, next_ch: int) -> str:
    """单角色弧光三行指令:当前值 + 当前拍→下一拍 + 禁止事项。"""
    lie = anchors.get("lie") or "—"
    at = anchors.get("arc_type") or "—"
    done = [b for b in arc_plan if b["target_chapter"] <= next_ch]
    ahead = [b for b in arc_plan if b["target_chapter"] > next_ch]
    cur = done[-1]["beat_name"] if done else "开场常态"
    nxt = (f"{ahead[0]['beat_name']}(目标第{ahead[0]['target_chapter']}章"
           + (f",信奉度趋 {ahead[0]['lie_belief_target']}" if ahead[0].get("lie_belief_target") is not None else "")
           + ")") if ahead else "终局"
    cur_v = f"{lie_belief}" if lie_belief is not None else "未测"
    return (f"【角色弧光·{name}】弧型:{at} | 信念(Lie):「{lie}」\n"
            f"当前信奉度:{cur_v};所处节拍:{cur} → 下一拍:{nxt}\n"
            f"本章指令:朝下一拍缓移,位移须由本章事件背书。"
            f"禁止:一步顿悟、借台词明说信念、跳到终局姿态。")


def promise_directives(rows: list[dict]) -> dict:
    """按窗口态分流:in_window→软提醒;over_half→简报指令(salience 低则先刷新);
    overdue→三选一决策。返回 {reminders, directives, decisions}(均为文案列表)。"""
    reminders, directives, decisions = [], [], []
    for p in rows:
        w = p.get("window") or {}
        state, gap = w.get("state"), w.get("salience_gap") or 0
        text = (p.get("text") or "")[:40]
        if state == "in_window":
            reminders.append(f"「{text}」在兑现窗口内,适时推进")
        elif state == "over_half":
            if gap > _SALIENCE_GAP_MAX:
                directives.append(
                    f"伏笔「{text}」已过窗口一半且久未露面({gap}章):本章先安排一次"
                    f"不动声色的刷新(换伪装:道具/闲笔/情绪),为兑现蓄势——先刷新,勿直接兑现")
            else:
                directives.append(f"伏笔「{text}」已过窗口一半:本章应实质推进或兑现")
        elif state == "overdue":
            decisions.append(
                f"伏笔「{text}」已超兑现窗口:请作者三选一——立即兑现 / 显式展期(改窗口) / "
                f"降级弃置(abandoned)。系统默认本章简报持续携带此项直到决策。")
    return {"reminders": reminders, "directives": directives, "decisions": decisions}


def open_loop_ledger(rows: list[dict], cap: int = _LOOP_CAP) -> dict:
    """开环账本:tier 加权计数(book×3/arc×2/chapter×1),只计窗口态非 inactive 的 open 线。"""
    weighted = sum(_TIER_W.get(p.get("tier") or "chapter", 1)
                   for p in rows if (p.get("window") or {}).get("state") not in (None, "inactive"))
    return {"weighted": weighted, "cap": cap, "over_cap": weighted > cap}


def heartbeat_list(chars: list[dict], next_ch: int, gap: int = 3) -> list[dict]:
    """待激活名单:有 private_agenda 且距上次活跃超过 gap 章的角色。"""
    out = []
    for c in chars:
        agenda = (c.get("private_agenda") or "").strip()
        last = c.get("last_active")
        if not agenda or last is None:
            continue
        if next_ch - last > gap:
            out.append({"name": c["name"], "agenda": agenda, "last_active": last})
    return out


def render_brief(sections: dict) -> str:
    parts = ["## 本章创作简报(章前组装,遵守但不复述)"]
    parts += list(sections.get("arcs") or [])
    pd = sections.get("promises") or {}
    if pd.get("directives"):
        parts.append("【伏笔·本章指令】\n" + "\n".join(f"- {x}" for x in pd["directives"]))
    if pd.get("decisions"):
        parts.append("【伏笔·待作者决策(阻塞级)】\n" + "\n".join(f"- {x}" for x in pd["decisions"]))
    if pd.get("reminders"):
        parts.append("【伏笔·在窗提醒】" + "；".join(pd["reminders"]))
    led = sections.get("loops")
    if led:
        warn = "——已超上限:先收一条再开新坑" if led["over_cap"] else ""
        parts.append(f"【开环账本】加权开环 {led['weighted']}/{led['cap']}{warn}")
    hb = sections.get("heartbeat") or []
    if hb:
        items = "；".join(f"{h['name']}(上次第{h['last_active']}章,其日程:{h['agenda']})" for h in hb)
        parts.append(f"【待激活角色】{items}——本章至少让其日程推进一步或被提及")
    return "\n\n".join(parts)


def assemble_chapter_brief(st, seq: int, focalizer: str = "") -> dict:
    """连 store 的薄壳:收集数据 → 各纯函数 → {sections, text}。seq 为即将写的章号。"""
    next_ch = seq
    max_ch = max(0, seq - 1)
    rows = st.promises_with_lifecycle(max_chapter=max_ch)
    arcs = []
    chars_hb = []
    for c in st.characters():
        a = st.char_anchors(c.id)
        states = st.list_char_states(c.id)
        last_active = max((s.get("chapter") or 0) for s in states) if states else None
        chars_hb.append({"id": c.id, "name": c.name,
                         "private_agenda": a.get("private_agenda"),
```

```python
def detect_mentions(text: str, entities: list[dict]) -> set[str]:
    """本章草料文本(synopsis+beats+演绎)中按 名字/别名 包含匹配命中的实体 id 集。"""
    hit = set()
    t = text or ""
    for e in entities:
        names = [e.get("name") or ""] + list(e.get("aliases") or [])
        if any(n and n in t for n in names):
            hit.add(e["id"])
    return hit


def select_injection(items: list[dict], mentioned_entities: set[str],
                     budget_chars: int = 1200) -> dict:
    """三态注入 + 预算裁剪。item: {id, entity, text, mode(always/detected/never), priority}。
    always 恒进;detected 须实体被提及;never 恒不进。超预算按 (always优先, priority 降序) 裁。"""
    def _mode(i):
        return i.get("mode") or "detected"
    pool = [i for i in items if _mode(i) == "always"] + sorted(
        [i for i in items if _mode(i) == "detected" and i.get("entity") in mentioned_entities],
        key=lambda i: -(i.get("priority") or 0))
    chosen, used, dropped = [], 0, 0
    for i in pool:
        cost = len(i.get("text") or "")
        if used + cost > budget_chars and _mode(i) != "always":
            dropped += 1
            continue
        chosen.append(i)
        used += cost
    return {"chosen": chosen, "dropped": dropped, "used_chars": used}
```

```python
def tension_deterministic(st, seq: int) -> list[dict]:
    """章后确定性张力项:进展路标 / 信息差盘点 / 开环账本。每项 {item, pass, detail, fix}。"""
    rows = st.promises_with_lifecycle(max_chapter=seq)
    touched = [p for p in rows if p.get("last_refresh_at") == seq]
    open_active = [p for p in rows if (p.get("window") or {}).get("state") not in (None, "inactive")]
    items = [{
        "item": "promise_progress",
        "pass": bool(touched) or not open_active,
        "detail": f"本章推进伏笔 {len(touched)} 条 / 开着 {len(open_active)} 条",
        "fix": "下一章至少让一条 open 伏笔产生路标级进展(关键信息/新障碍/关键决定)",
    }]
    led = open_loop_ledger(rows)
    items.append({"item": "open_loops", "pass": not led["over_cap"],
                  "detail": f"加权开环 {led['weighted']}/{led['cap']}",
                  "fix": "先收一条再开新坑"})
    # 信息差盘点:读者知而在场角色不知的命题数(irony 面)。空 → 提示主动制造。
    try:
        irony = st.dramatic_irony(seq) if hasattr(st, "dramatic_irony") else []
    except Exception:
        irony = []
    items.append({"item": "irony_surface", "pass": bool(irony),
                  "detail": f"读者知/角色不知命题 {len(irony)} 条",
                  "fix": "考虑向读者(而非角色)提前揭示某高危信息,把后续普通戏变成悬念戏"})
```

```python
def tension_check(seq: int, work: str = "main"):
```
