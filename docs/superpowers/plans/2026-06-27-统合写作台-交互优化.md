```python
    def add_chapter_version(self, seq: int, prose: str, label: str = "",
                            kind: str = "checkpoint", qc: dict | None = None) -> str:
        """记一个命名存档点。返回新版本的裸 id。"""
        rows = self._query(
            "CREATE chapter_version SET seq=$seq, prose=$prose, label=$label, "
            "kind=$kind, qc=$qc, created_at=time::now()",
            {"seq": seq, "prose": prose, "label": label, "kind": kind, "qc": qc or {}})
        # CREATE 返回新记录(含 id,形如 'chapter_version:xxxx');取裸 id 部分。
        rid = rows[0]["id"] if rows else None
        return str(rid).split(":")[-1] if rid else ""

    def list_chapter_versions(self, seq: int) -> list[dict]:
        """某章存档点列表(不含 prose 全文),最新在前。"""
        return self._query(
            "SELECT meta::id(id) AS id, seq, label, kind, qc, created_at "
            "FROM chapter_version WHERE seq=$seq ORDER BY created_at DESC",
            {"seq": seq})

    def get_chapter_version(self, vid: str) -> dict | None:
        """单个存档点全文(回看用)。"""
        rows = self._query(
            "SELECT meta::id(id) AS id, seq, prose, label, kind, qc, created_at "
            "FROM type::record('chapter_version', $vid)", {"vid": vid})
        return rows[0] if rows else None
```

// 把主编提案应用到正文。selection 用选区偏移精确替换;原文漂移则拒绝(不静默追加)。
// 返回 { next, ok, drifted? }。
export function applyProposal(prose, proposal, sel) {
  const p = proposal?.prose ?? ''
  if (proposal?.scope === 'chapter') return { next: p, ok: true }
  if (proposal?.scope === 'append') {
    return { next: (prose ? prose + '\n\n' : '') + p, ok: true }
  }
  // selection:需要有效选区偏移
  if (sel && Number.isInteger(sel.from) && Number.isInteger(sel.to)) {
    if (prose.slice(sel.from, sel.to) === sel.text) {
      return { next: prose.slice(0, sel.from) + p + prose.slice(sel.to), ok: true }
    }
    return { next: prose, ok: false, drifted: true }   // 漂移:交给 UI 提示作者
  }
  // 无偏移信息:退化成追加(不猜测位置)
  return { next: (prose ? prose + '\n\n' : '') + p, ok: true }
}
```

// 可复用改写指令按钮:按 work 隔离,存 localStorage(改造⑤)。
const key = (work) => `scribe3:rewriteBtns:${work}`
export function loadButtons(work) {
  try { return JSON.parse(localStorage.getItem(key(work)) || '[]') }
  catch { return [] }
}
export function addButton(work, label) {
  const t = (label || '').trim()
  if (!t) return loadButtons(work)
  const cur = loadButtons(work)
  if (cur.includes(t)) return cur
  const next = [...cur, t]
  localStorage.setItem(key(work), JSON.stringify(next))
  return next
}
export function removeButton(work, label) {
  const next = loadButtons(work).filter((x) => x !== label)
  localStorage.setItem(key(work), JSON.stringify(next))
  return next
}
```

```jsx
import { useEffect, useState } from 'react'
import { api } from './api'
import { Empty } from './ui'

// 知识帧 / 戏剧反讽 / 北极星意图(原 DeduceLayout 右面板,迁入写作台左面板)。
export function FramePanel({ focalizer, chapter }) {
  const [frame, setFrame] = useState([])
  const [irony, setIrony] = useState([])
  const [intent, setIntent] = useState({ reader_feeling: '', emotional_arc: '', red_lines: '' })
  useEffect(() => {
    if (focalizer) {
      api.frame(focalizer, chapter).then(setFrame).catch(() => setFrame([]))
      api.irony(focalizer, chapter).then(setIrony).catch(() => setIrony([]))
    } else { setFrame([]); setIrony([]) }
  }, [focalizer, chapter])
  useEffect(() => { api.intent().then(setIntent).catch(() => {}) }, [])
  return (
    <div className="space-y-4">
      <section>
        <div className="text-[12px] font-medium text-ink mb-1.5">👁 知识帧</div>
        {!focalizer ? <Empty>作者全知视角——无限知约束</Empty>
          : frame.length === 0 ? <Empty>该角色此章暂无已知事实</Empty>
          : <ul className="space-y-1">{frame.map((f) => <li key={f.id} className="text-[12px] text-ink">{f.value || f.id}</li>)}</ul>}
      </section>
      <section>
        <div className="text-[12px] font-medium text-ink mb-1.5">🎭 戏剧反讽(读者知道,他不知道)</div>
        {!focalizer ? <Empty>切某角色视角后显示</Empty>
          : irony.length === 0 ? <Empty>无</Empty>
          : <ul className="space-y-1">{irony.map((f) => <li key={f.id} className="text-[12px] text-faint">{f.value || f.id}</li>)}</ul>}
      </section>
      <section>
        <div className="text-[12px] font-medium text-ink mb-1.5">🎯 北极星意图</div>
        {['reader_feeling', 'emotional_arc', 'red_lines'].map((k) => (
          <textarea key={k} value={intent[k]} placeholder={k}
            onChange={(e) => setIntent((s) => ({ ...s, [k]: e.target.value }))}
            className="w-full mb-1.5 rounded-lg border border-line bg-canvas px-2 py-1 text-[12px] text-ink outline-none focus:border-brand" rows={2} />
        ))}
        <button onClick={() => api.putIntent(intent)} className="text-[11px] text-brand hover:bg-soft px-2 h-6 rounded-lg cursor-pointer">保存意图</button>
      </section>
    </div>
  )
}
```

在 WriteLayout 左面板 tab 列表加一个 tab(如「📐 帧/意图」)渲染 `<FramePanel focalizer={focalizer} chapter={chapter} />`。删 DeduceLayout 时其逻辑已被本组件覆盖。
