```surql
-- 声音库:每角色 4 类独白(自由/压力/关系/叙述),可版本化(同 character+kind 追加新 since_chapter)。
DEFINE TABLE IF NOT EXISTS voice_sample SCHEMALESS;
DEFINE FIELD IF NOT EXISTS character ON voice_sample TYPE string;
DEFINE FIELD IF NOT EXISTS kind ON voice_sample TYPE string;          -- free|pressure|relation|narration
DEFINE FIELD IF NOT EXISTS text ON voice_sample TYPE string;
DEFINE FIELD IF NOT EXISTS since_chapter ON voice_sample TYPE int;

-- 声音进化标记:ingest 检测重大事件 → 标记(只提示,不自动生成独白)。
DEFINE TABLE IF NOT EXISTS voice_evolution_flag SCHEMALESS;
DEFINE FIELD IF NOT EXISTS character ON voice_evolution_flag TYPE string;
DEFINE FIELD IF NOT EXISTS seq ON voice_evolution_flag TYPE int;
DEFINE FIELD IF NOT EXISTS reason ON voice_evolution_flag TYPE string;
DEFINE FIELD IF NOT EXISTS status ON voice_evolution_flag TYPE string;  -- open|done
```

- [ ] **Step 3b: store.py 追加(在 craft_rule 方法之后)**

```python
    # --- 声音库(per-character 4类独白 + 版本)---
    def add_voice_sample(self, character: str, kind: str, text: str, since_chapter: int = 0) -> None:
        self._execute_write(
            "CREATE voice_sample SET character=$c, kind=$k, text=$t, since_chapter=$s",
            {"c": character, "k": kind, "t": text, "s": since_chapter})

    def list_voice_samples(self, character: str) -> list[dict]:
        return self._query(
            "SELECT meta::id(id) AS id, character, kind, text, since_chapter "
            "FROM voice_sample WHERE character=$c ORDER BY since_chapter", {"c": character})

    def latest_voice_samples(self, character: str, upto_chapter: int) -> dict:
        """时间门:每 kind 取 since_chapter≤upto 的最新版(不剧透未来版本)。返回 {kind: text}。"""
        rows = self._query(
            "SELECT kind, text, since_chapter FROM voice_sample "
            "WHERE character=$c AND since_chapter <= $u ORDER BY since_chapter",
            {"c": character, "u": upto_chapter})
        out: dict = {}
        for r in rows:                 # 升序遍历,后覆盖前 → 各 kind 最新版
            out[r["kind"]] = r["text"]
        return out

    # --- 声音进化标记 ---
    def add_voice_flag(self, character: str, seq: int, reason: str) -> None:
        self._execute_write(
            "CREATE voice_evolution_flag SET character=$c, seq=$s, reason=$r, status='open'",
            {"c": character, "s": seq, "r": reason})

    def list_voice_flags(self, status: str = "open") -> list[dict]:
        return self._query(
            "SELECT meta::id(id) AS id, character, seq, reason, status "
            "FROM voice_evolution_flag WHERE status=$st", {"st": status})
```

```python
def parse_knowledge_delta(reply: str) -> dict:
    try:
        m = re.search(r"\{.*\}", reply or "", re.S)
        d = json.loads(m.group(0)) if m else {}
        return {"learned": list(d.get("learned") or []), "did": list(d.get("did") or []),
                "voice_shift": list(d.get("voice_shift") or [])}
    except Exception:
        return {"learned": [], "did": [], "voice_shift": []}
```
