```python
class LlmStub:
    """可编程 LLM 边界替身。默认正常回流;可注入故障测 fail-loud;记录每次调用(含 history)。"""

    def __init__(self):
        self._reply_deltas = ["好", "的"]
        self._exc = None
        self._events = []
        self.calls = []  # 每次 LLM 调用的 kwargs 快照,含 history/system

    def reply(self, text):
        """设定下一次流式回流的文本(按字符切成 deltas)。"""
        self._reply_deltas = list(text)
        self._exc = None

    def fail(self, exc):
        """设定下一次 LLM 调用抛出的异常(测 fail-loud)。"""
        self._exc = exc

    def responses(self, events):
        """设定 agentic 事件流(hermes_responses_events 的产出)。"""
        self._events = list(events)
        self._exc = None

    def complete_stream(self, msg, system=None, history=None, **kw):
        self.calls.append({"msg": msg, "system": system, "history": history, **kw})
        if self._exc:
            raise self._exc
        return iter(self._reply_deltas)

    def responses_events(self, message, instructions, **kwargs):
        self.calls.append({"message": message, "instructions": instructions, **kwargs})
        if self._exc:
            raise self._exc
        return iter(self._events)

    @property
    def last_history(self):
        return self.calls[-1].get("history") if self.calls else None
```

```python
"""跨测试复用的 SSE 解析 + 断言。只放真正复用两次以上的;单次断言内联在测试里。"""
import json


def sse_frames(response):
    """把 TestClient 的流式响应解析成 [dict, ...](每个 data: 行的 JSON)。"""
    body = b"".join(response.iter_bytes()).decode()
    frames = []
    for line in body.splitlines():
        line = line.strip()
        if line.startswith("data:"):
            payload = line[len("data:"):].strip()
            try:
                frames.append(json.loads(payload))
            except json.JSONDecodeError:
                pass
    return frames


def assert_error_frame_carries_cause(response, cause_substr):
    """断言 SSE 里有 error 帧且携带真实异常原文,并以 done 帧优雅收尾(非裸断/500)。"""
    frames = sse_frames(response)
    errs = [f for f in frames if "error" in f]
    assert errs, f"应有 error 帧,实际帧={frames!r}"
    assert any(cause_substr in str(f["error"]) for f in errs), \
        f"error 帧应含原文 {cause_substr!r},实际={errs!r}"
    assert any(f.get("done") for f in frames), f"应有 done 帧优雅收尾,实际={frames!r}"
```

```javascript
// 真机操作库:两档前端 e2e 共用。从 pw.mjs 提炼。
import { chromium } from 'playwright'

export const BASE = process.env.E2E_BASE || 'http://127.0.0.1:3311'
const sleep = (ms) => new Promise((r) => setTimeout(r, ms))

export async function withPage(fn) {
  const b = await chromium.launch()
  const p = await b.newPage({ viewport: { width: 1440, height: 900 } })
  const errs = []
  p.on('pageerror', (e) => errs.push('PAGEERR ' + e.message))
  p.on('console', (m) => {
    if (m.type() === 'error') {
      const t = m.text()
      if (!/ERR_CONNECTION_CLOSED|favicon/.test(t)) errs.push('CONSOLE ' + t.slice(0, 140))
    }
  })
  try { const r = await fn(p); return { ok: true, errs, ...r } }
  catch (e) { return { ok: false, err: e.message, errs } }
  finally { await b.close() }
}

export async function openBook(p, title) {
  await p.goto(BASE, { waitUntil: 'networkidle' }); await sleep(800)
  await p.getByText(title, { exact: false }).first().click()
  await sleep(2000)
  await p.waitForSelector('[role="treeitem"]', { timeout: 8000 }).catch(() => {})
}

export const visText = async (p) =>
  (await p.evaluate(() => document.body.innerText)).replace(/\s+/g, ' ').slice(0, 2000)
```
