```python
    def create_editorial_thread(self, work: str, actor_id: str,
                                title: str | None = None) -> dict:
        """显式新建主编线程(P0a 多会话):与 get_or_create 不同,每次都建新的。
        默认标题「新会话」——首条 user 消息落库时由 maybe_autotitle_thread 自动改名。"""
        from uuid import uuid4
        tid = f"ethread:{work}:{uuid4().hex}"
        self._execute_write(
            "CREATE editorial_thread SET thread_id=$tid, work=$w, actor_id=$a, "
            "parent_thread_id=NONE, title=$title, status='active', "
            "hermes_conversation_id=NONE, previous_response_id=NONE, summary='', "
            "updated_at=time::now() RETURN NONE",
            {"tid": tid, "w": work, "a": actor_id, "title": (title or "").strip() or "新会话"})
        return self._get_editorial_thread(tid)

    def update_editorial_thread_meta(self, thread_id: str, *,
                                     title: str | None = None,
                                     status: str | None = None) -> None:
        """改名/归档(P0a):只改传入字段。status 合法值 active|archived(调用方校验)。"""
        sets, params = ["updated_at=time::now()"], {"tid": thread_id}
        if title is not None:
            sets.append("title=$title"); params["title"] = title
        if status is not None:
            sets.append("status=$status"); params["status"] = status
        self._execute_write(
            f"UPDATE editorial_thread SET {', '.join(sets)} "
            "WHERE thread_id=$tid RETURN NONE", params)
```

```python
class ThreadCreateReq(BaseModel):
    title: str | None = None


class ThreadPatchReq(BaseModel):
    title: str | None = None
    status: str | None = None


@app.post("/chat/threads")
def chat_thread_create(req: ThreadCreateReq, work: str = "main", actor_id: str = "author"):
    """P0a:显式新建会话(与 GET /chat/threads 的幂等取默认不同,每次都建新线程)。"""
    if req.title is not None and len(req.title) > 60:
        raise HTTPException(status_code=422, detail="标题过长(≤60 字)")
    return _store_for(work).create_editorial_thread(work, actor_id, title=req.title)


@app.patch("/chat/threads/{thread_id}")
def chat_thread_patch(thread_id: str, req: ThreadPatchReq, work: str = "main"):
    """P0a:改名/归档。零降级:线程不存在 404、跨 work 409、非法 status 422。"""
    if req.status is not None and req.status not in ("active", "archived"):
        raise HTTPException(status_code=422, detail=f"非法 status={req.status!r}(active|archived)")
    if req.title is not None and len(req.title) > 60:
        raise HTTPException(status_code=422, detail="标题过长(≤60 字)")
    _assert_work_binding(_thread_work(thread_id), work,
                         kind="editorial_thread", ident=thread_id)
    st = _store_for(work)
    if st._get_editorial_thread(thread_id) is None:
        raise HTTPException(status_code=404, detail={
            "code": "thread_not_found", "message": f"线程 '{thread_id}' 不存在"})
    st.update_editorial_thread_meta(thread_id, title=req.title, status=req.status)
    return st._get_editorial_thread(thread_id)
```

```python
def _resolve_thread_or_raise(work: str, thread: str | None):
    """P0a 零降级:thread 传了必须有效——不存在 404 / 已归档 409 / 跨 work 409。
    绝不静默回落 active(回落=串台掩盖器)。返回线程行或 None(未传)。"""
    if not thread:
        return None
    _assert_work_binding(_thread_work(thread), work,
                         kind="editorial_thread", ident=thread)
    row = _store_for(work)._get_editorial_thread(thread)
    if row is None:
        raise HTTPException(status_code=404, detail={
            "code": "thread_not_found", "message": f"线程 '{thread}' 不存在"})
    if row.get("status") == "archived":
        raise HTTPException(status_code=409, detail={
            "code": "thread_archived", "message": "会话已归档,恢复(PATCH status=active)后再用"})
    return row
```

```jsx
// P0a:主编会话下拉——列表/新建/切换/改名/归档/分支。仅 threadMode(有 snapshot)时由父级渲染。
import { useEffect, useRef, useState } from 'react'
import { ChevronDown, Plus, Pencil, Archive, GitBranch, Download } from 'lucide-react'
import { api } from './api'
import { useContextSnapshot } from './context/ContextSnapshotContext'
import { showToast } from './ui'   // 按 studioDocks.jsx 实际 import 路径对齐

export function ThreadSwitcher({ busy }) {
  const { snapshot, switchThread } = useContextSnapshot()
  const [open, setOpen] = useState(false)
  const [threads, setThreads] = useState(null)   // null=未加载;加载失败显式 error 态
  const [error, setError] = useState(false)
  const boxRef = useRef(null)
  const current = snapshot?.thread_id || null
  const pull = () => api.threads()
    .then((d) => { setThreads(d.threads || []); setError(false) })
    .catch(() => { setThreads([]); setError(true) })   // fail-loud:error 态红字,不装没事
  useEffect(() => { if (open) pull() }, [open])
  useEffect(() => {
    if (!open) return
    const onDown = (e) => { if (boxRef.current && !boxRef.current.contains(e.target)) setOpen(false) }
    document.addEventListener('mousedown', onDown)
    return () => document.removeEventListener('mousedown', onDown)
  }, [open])

  const guardBusy = () => !busy
    || window.confirm('主编正在生成:切换会话将中止本轮(已生成部分已留痕)。继续?')
  const doSwitch = (tid) => { if (tid === current || !guardBusy()) return; setOpen(false); switchThread(tid) }
  const doNew = async () => {
    if (!guardBusy()) return
    try { const t = await api.createThread(); setOpen(false); switchThread(t.thread_id) }
    catch (e) { showToast(`新建会话失败:${e?.body?.detail || e?.message}`, { tone: 'danger' }) }
  }
  const doRename = async (t) => {
    const name = window.prompt('会话改名', t.title || '')
    if (name == null) return
    try { await api.patchThread(t.thread_id, { title: name }); pull() }
    catch (e) { showToast(`改名失败:${e?.body?.detail || e?.message}`, { tone: 'danger' }) }
  }
  const doArchive = async (t) => {
    try {
      await api.patchThread(t.thread_id, { status: 'archived' }); pull()
      if (t.thread_id === current) {   // 归档当前 → 显式切默认(active 首条),toast 告知
        const d = await api.threads()
        const next = (d.threads || []).find((x) => x.status === 'active')
        if (next) { switchThread(next.thread_id); showToast('当前会话已归档,已切到默认会话', { tone: 'info' }) }
      }
    } catch (e) { showToast(`归档失败:${e?.body?.detail || e?.message}`, { tone: 'danger' }) }
  }
  const doFork = async () => {
    if (!current) return
    try { const t = await api.forkThread(current, {}); setOpen(false); switchThread(t.thread_id) }
    catch (e) { showToast(`分支失败:${e?.body?.detail || e?.message}`, { tone: 'danger' }) }
  }

  const active = (threads || []).filter((t) => t.status === 'active')
  const archived = (threads || []).filter((t) => t.status === 'archived')
  const curTitle = (threads || []).find((t) => t.thread_id === current)?.title || '主编讨论'
  return (
    <div ref={boxRef} className="relative shrink-0">
      <button onClick={() => setOpen((v) => !v)}
        className="flex items-center gap-1 text-f12 text-ink hover:text-brand cursor-pointer max-w-[220px]">
        <span className="truncate">{curTitle}</span><ChevronDown size={12} />
      </button>
      {open && (
        <div className="absolute z-40 mt-1 w-64 rounded-lg border border-line bg-surface shadow-lg p-1">
          {threads == null && <div className="px-2 py-1.5 text-f12 text-faint">加载中…</div>}
          {error && <div className="px-2 py-1.5 text-f12 text-danger">会话列表加载失败(不代表没有会话)</div>}
          {active.map((t) => (
            <div key={t.thread_id} className={`group flex items-center gap-1 px-2 py-1.5 rounded hover:bg-soft ${t.thread_id === current ? 'text-brand' : 'text-ink'}`}>
              <button onClick={() => doSwitch(t.thread_id)} className="flex-1 text-left text-f12 truncate cursor-pointer">
                {t.thread_id === current ? '✓ ' : ''}{t.title || '未命名'}
              </button>
              <button title="改名" onClick={() => doRename(t)} className="opacity-0 group-hover:opacity-100"><Pencil size={11} /></button>
              <button title="归档" onClick={() => doArchive(t)} className="opacity-0 group-hover:opacity-100"><Archive size={11} /></button>
            </div>
          ))}
          <div className="border-t border-line/60 mt-1 pt-1 space-y-0.5">
            <button onClick={doNew} className="w-full flex items-center gap-1.5 px-2 py-1.5 text-f12 text-brand hover:bg-soft rounded cursor-pointer"><Plus size={12} />新建会话</button>
            <button onClick={doFork} className="w-full flex items-center gap-1.5 px-2 py-1.5 text-f12 text-faint hover:bg-soft rounded cursor-pointer"><GitBranch size={12} />从当前分支(带摘要,消息从空开始)</button>
          </div>
          {archived.length > 0 && (
            <div className="border-t border-line/60 mt-1 pt-1">
              <div className="px-2 py-0.5 text-meta text-faint">已归档 {archived.length}</div>
              {archived.map((t) => (
                <div key={t.thread_id} className="flex items-center gap-1 px-2 py-1 text-f12 text-faint">
                  <span className="flex-1 truncate">{t.title}</span>
                  <button className="text-brand cursor-pointer" onClick={async () => {
                    try { await api.patchThread(t.thread_id, { status: 'active' }); pull() }
                    catch (e) { showToast(`恢复失败:${e?.message}`, { tone: 'danger' }) }
                  }}>恢复</button>
                </div>
              ))}
            </div>
          )}
        </div>
      )}
    </div>
  )
}
```
